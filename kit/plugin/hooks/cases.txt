# Gate regression cases - run by check-gates.sh. Format: a header line `### <role> <expected>`
# followed by the command (one or more lines) up to the next header. Lines starting with # are comments.
#   role      interactive | overseer | worker | overlord   (AGENT_ROLE; interactive = unset)
#   expected  allow | ask:H4 | block:H1 | block:H2 | block:H4
#             ask:H4 = guard-bash.sh exits 0 with a permissionDecision "ask" (attended sessions: the tap is the go)
#   (optional 4th word) tool = feed the text to guard-tools.sh (MCP/WebFetch H1) instead of guard-bash.sh;
#                       bypass = send permission_mode=bypassPermissions in the hook input (an ask must become a block);
#                       fp = a documented, accepted false positive: the case pins today's behaviour (runs as a plain case)
# Config-safety tests (config never executed, malformed config inert, case-insensitive H1, session id
# validation) are built into check-gates.sh because they need their own config files.
# check-gates.sh builds a temp project: estate.env with SECRET_CLI=secretcli, guard.conf with
# H1_PATTERNS='forbidden-target', H4_PROVIDER_HOSTS='api.provider.example', H4_NEVER_PATHS='/data/photos|_keep'
# and H4_ASK_PATTERNS='deploy-to-prod', and sets AGENT_EVAL=1.
# The two lists: the your-hand shapes ASK in an attended session (role interactive) and BLOCK in an unattended
# one; H1, H2 and H4_NEVER_PATHS BLOCK in every session; branch discipline (git push, main) binds unattended only.
# Whenever a real block or a false positive happens, add it here in the same change as the hook edit.

cat > /tmp/x.sh <<'EOF'
sudo reboot
EOF
bash /tmp/x.sh
### interactive ask:H4
cat > /tmp/x.sh <<'EOF'
sudo reboot
EOF
chmod +x /tmp/x.sh; /tmp/x.sh
### worker block:H4
cat > /tmp/x.sh <<'EOF'
sudo reboot
EOF
bash /tmp/x.sh
# prose writes still pass in an attended session (one opener line, any body of plain text)
### interactive allow
cat <<-EOF > notes.md
	sudo reboot is the owner's tap
	EOF
### interactive allow
tee -a notes.md >/dev/null <<EOF
the host will reboot
EOF
### interactive allow
 # a note for the brief

# every block reads the full text: heredoc prose that itself holds a redirect into a control file blocks (H7)
### interactive block:H7 fp
cat > docs/notes.md <<'EOF'
reset it with: echo '{}' > agents/state/ceiling.json
EOF
### worker block:H7
cat > docs/notes.md <<'EOF'
echo x >> agents/bin/tick
EOF
# accepted false positive: an opener whose FILE word is quoted or holds a variable is not on the allowlist, so prose
# written through it is judged on the full text (fail closed); use a plain file name to keep the strip
### interactive ask:H4 fp
cat > "$HOME/notes.md" <<'EOF'
sudo reboot is the owner's tap
EOF

# ---- the grep-pattern blanking never starts at a quote that is literal inside the OTHER kind of quoting ------------
# the apostrophe after "see grep" is plain text inside the double quotes, so the reboot after the ; is a real command
### overseer block:H4
echo "see grep 'x"; sudo reboot; echo 'y'
### interactive ask:H4
echo "see grep 'x"; sudo reboot; echo 'y'
# ... and the mirror image: a double quote that is plain text inside single quotes
### overseer block:H4
echo 'see grep "x'; sudo reboot; echo "y"
### interactive ask:H4
echo 'see grep "x'; sudo reboot; echo "y"
# a pattern that itself holds the other quote character is simply not blanked: a false-positive ask at worst
### interactive ask:H4 fp
grep -E "don't|shutdown" docs/runbook.md
# real quoted patterns are still blanked
### interactive allow
grep -E 'battery|shutdown' docs/runbook.md
### overseer allow
grep -E "battery|shutdown" docs/runbook.md
